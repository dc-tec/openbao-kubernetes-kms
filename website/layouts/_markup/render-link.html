{{- $dest := .Destination -}} {{- $parts := split $dest "#" -}} {{- $pathPart := index $parts 0 -}} {{- $fragment := "" -}} {{- if gt (len $parts) 1 -}}{{- $fragment = index $parts 1 -}}{{- end -}} {{- $href := $dest -}} {{- $isExternal := or (hasPrefix $pathPart "http://") (hasPrefix $pathPart "https://") (hasPrefix $pathPart "mailto:") (hasPrefix $pathPart "tel:") -}} {{- if and (not (hasPrefix $dest "#")) (not $isExternal) -}} {{- $resolved := $pathPart -}} {{- if and (strings.HasSuffix $pathPart ".md") (not (hasPrefix $pathPart "/")) -}} {{- $resolved = path.Clean (path.Join (path.Dir .Page.File.Path) $pathPart) -}} {{- end -}} {{- $targetSite := .Page.Site -}} {{- if and (eq .Page.Section "contribute") (hasPrefix $resolved "/docs/") -}} {{- range hugo.Sites -}}{{- if eq .Version.Name "next" -}}{{- $targetSite = . -}}{{- end -}}{{- end -}} {{- end -}} {{- if hasPrefix (strings.TrimPrefix "/" $resolved) "contribute/" -}} {{- $targetSite = hugo.Sites.Default -}} {{- end -}} {{- with $targetSite.GetPage $resolved -}} {{- $href = .RelPermalink -}} {{- else -}} {{- if or (hasPrefix $pathPart "./") (hasPrefix $pathPart "../") -}} {{- $line := index hugo.Data.version_lines $.Page.Site.Version.Name -}} {{- $ref := cond (eq $line.sourceRef "HEAD") $.Page.Site.Params.defaultBranch $line.sourceRef -}} {{- $repoBase := $.Page.File.Path -}} {{- if eq $.Page.Section "contribute" -}} {{- $ref = $.Page.Site.Params.defaultBranch -}} {{- $repoBase = path.Join "docs" $repoBase -}} {{- end -}} {{- $repoPath := path.Clean (path.Join (path.Dir $repoBase) $pathPart) -}} {{- $href = printf "%s/blob/%s/%s" $.Page.Site.Params.repository $ref $repoPath -}} {{- else if hasPrefix $pathPart "/" -}} {{- $href = relURL (strings.TrimPrefix "/" $pathPart) -}} {{- end -}} {{- end -}} {{- if $fragment -}}{{- $href = printf "%s#%s" $href $fragment -}}{{- end -}} {{- end -}} {{ .Text | safeHTML }}
